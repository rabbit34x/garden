## 確認用

```
IMAGE_NAME=""
IMAGE_TAG="develop"

DEVELOP_CONTAINER_ID=$(sudo docker ps -a | grep ${IMAGE_NAME} | grep ${IMAGE_TAG} | awk '{print $1}')
DEVELOP_IMAGE_ID=$(sudo docker images | grep ${IMAGE_NAME} | grep ${IMAGE_TAG} | awk '{print $3}')

echo ${DEVELOP_CONTAINER_ID}
echo ${DEVELOP_IMAGE_ID}
```


## 実行

```
IMAGE_NAME=""
IMAGE_TAG="develop"

DEVELOP_CONTAINER_ID=$(sudo docker ps -a | grep ${IMAGE_NAME} | grep ${IMAGE_TAG} | awk '{print $1}')
DEVELOP_IMAGE_ID=$(sudo docker images | grep ${IMAGE_NAME} | grep ${IMAGE_TAG} | awk '{print $3}')

sudo docker stop ${DEVELOP_CONTAINER_ID}
sudo docker rm ${DEVELOP_CONTAINER_ID}
sudo docker rmi ${DEVELOP_IMAGE_ID}
```
